{%- comment -%} Formats a date range for CV badges. Parameters: include.start — startDate (YYYY or YYYY-MM) include.end — endDate (YYYY or YYYY-MM, may be empty) include.present — true to show "– Present" when end is empty Examples: "2022-09" / "2022-12" → Sep–Dec 2022 "2020-09" / "2022-06" → Sep 2020 – Jun 2022 "2022-09" / "" → Sep 2022 "2022-09" / "" present=true → Sep 2022 – Present "2022" / "2024" → 2022–2024 {%- endcomment -%} {%- assign _months = "Jan,Feb,Mar,Apr,May,Jun,Jul,Aug,Sep,Oct,Nov,Dec" | split: "," -%} {%- comment -%}— parse start —{%- endcomment -%} {%- assign _sp = include.start | split: "-" -%} {%- assign _sy = _sp[0] -%} {%- assign _splen = _sp | size -%} {%- if _splen > 1 -%} {%- assign _smi = _sp[1] | plus: 0 | minus: 1 -%} {%- assign _sm = _months[_smi] -%} {%- else -%} {%- assign _sm = nil -%} {%- endif -%} {%- comment -%}— build start label —{%- endcomment -%} {%- if _sm -%} {%- assign _slabel = _sm | append: " " | append: _sy -%} {%- else -%} {%- assign _slabel = _sy -%} {%- endif -%} {%- if include.end and include.end != "" -%} {%- comment -%}— parse end —{%- endcomment -%} {%- assign _ep = include.end | split: "-" -%} {%- assign _ey = _ep[0] -%} {%- assign _eplen = _ep | size -%} {%- if _eplen > 1 -%} {%- assign _emi = _ep[1] | plus: 0 | minus: 1 -%} {%- assign _em = _months[_emi] -%} {%- else -%} {%- assign _em = nil -%} {%- endif -%} {%- comment -%}— same year: collapse year —{%- endcomment -%} {%- if _sy == _ey -%} {%- if _sm and _em -%} {%- assign _out = _sm | append: "–" | append: _em | append: " " | append: _sy -%} {%- else -%} {%- assign _out = _slabel -%} {%- endif -%} {%- else -%} {%- comment -%}— cross-year —{%- endcomment -%} {%- if _em -%} {%- assign _elabel = _em | append: " " | append: _ey -%} {%- else -%} {%- assign _elabel = _ey -%} {%- endif -%} {%- assign _out = _slabel | append: " – " | append: _elabel -%} {%- endif -%} {%- elsif include.present -%} {%- assign _out = _slabel | append: " – Present" -%} {%- else -%} {%- assign _out = _slabel -%} {%- endif -%} {{- _out -}}
